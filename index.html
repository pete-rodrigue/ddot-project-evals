what we’ll do:

  • get 10 meter buffer around every individual road segment

  • count the number of crashes in each buffered segment in each year

    • if crashes fall into multiple buffers, assign them to a single buffer randomly. this adds noise and will attenuate the “treatment effect”

  • merge on which segments had PBLs & buffered bike lanes installed

  • run regression that compares crash numbers per year, controlling for area, pre-install crashes, road width, bike traffic from Cabi system, etc

  • compare simple before and after means on PBL/BBL & non-PBL/BBL segments

run_regs <- function(df, outcome, include_year = TRUE) {
  root_reg <- paste0(outcome, " ~ treatment + post_install + treatment*post_install")
  year_term <- ifelse(include_year, " + as.factor(year)", "")

  formulas <- list(
    simple = as.formula(root_reg),
    full   = as.formula(paste0(
      root_reg,
      " + mean_crashes_pre2019 + TOTALTRAVELLANEWIDTH + TOTALPARKINGLANES",
      " + as.factor(WARD_ID) + as.factor(DCFUNCTIONALCLASS) + total_change_trips",
      year_term
    ))
  )

  map_dfr(formulas, \(f) {
    tidy(lm(f, data = df)) |>
      filter(term == "treatment:post_install") |>
      mutate(outcome = outcome)
  }, .id = "model")
}


plot_effects <- function(outcomes) {
  results <- map_dfr(outcomes, \(o) {
    run_regs(df = cd_pbl_clean_simplified, outcome = o, include_year = FALSE)
  })
  
  results <- results |>
    mutate(
      significance = case_when(
        p.value < 0.001 ~ "p < 0.001",
        p.value < 0.01  ~ "p < 0.01",
        p.value < 0.05  ~ "p < 0.05",
        TRUE            ~ "p ≥ 0.05"
      ),
      significance = factor(significance,
                            levels = c("p < 0.001", "p < 0.01",
                                       "p < 0.05",  "p ≥ 0.05")),
      model  = factor(model, levels = c("simple", "full"),
                      labels = c("Simple model", "Full model")),
      outcome = factor(outcome, levels = outcomes) |> # preserve ordering
        fct_relabel(\(x) str_replace_all(x, "_", " ") |>
                                str_replace(" per acre", "") |>
                                str_to_sentence())
    )
  
  
  sig_colors <- c(
    "p < 0.001" = "#2166ac",
    "p < 0.01"  = "#74add1",
    "p < 0.05"  = "#fdae61",
    "p ≥ 0.05"  = "#d9d9d9"
  )
  
  ggplot(
         filter(results, model=="Simple model"), 
         aes(x = outcome, y = estimate, fill = significance)) +
    geom_col(position = position_dodge(width = 0.7), width = 0.6) +
    geom_errorbar(
      aes(ymin = estimate - 1.96 * std.error,
          ymax = estimate + 1.96 * std.error),
      position = position_dodge(width = 0.7),
      width = 0.2, color = "grey30"
    ) +
    geom_hline(yintercept = 0, color = "black", linewidth = 0.5) +
    scale_fill_manual(values = sig_colors, drop = FALSE) +
    scale_x_discrete(labels = \(x) str_replace_all(x, "_", "\n")) +
    # facet_wrap(~model) +
    labs(
      title = "Effect of protected bike lane installation on crash outcomes",
      subtitle = "Coefficient on treatment × post-install interaction",
      x     = NULL,
      y     = "Estimated effect per acre",
      fill  = "Significance"
    ) +
    theme_minimal() +
    theme(
      axis.text.x     = element_text(size = 8),
      legend.position = "bottom",
      panel.grid.major.x = element_blank()
    )
}

      "total_vehicles_per_acre",

  outcomes = c(
      "crashes_per_acre",
      "injury_crashes_per_acre",
      
      "driver_injury_crashes_per_acre",
      "serious_driver_injury_crashes_per_acre",
  
      "bike_injury_crashes_per_acre",
      "serious_bike_injury_crashes_per_acre",
      
      "ped_injury_crashes_per_acre",
      "serious_ped_injury_crashes_per_acre"
      )

summary_df <-
  cd_pbl_clean_simplified |>
  select(SUBBLOCKKEY, post_install, treatment, ends_with("_per_acre")) |>
  pivot_longer(cols = all_of(outcomes),
               names_to  = "outcome",
               values_to = "value") |>
  group_by(outcome, post_install, treatment) |>        # add treatment to grouping
  summarise(
    mean = mean(value, na.rm = TRUE),
    .groups = "drop"
  ) |>
  mutate(
    post_install = factor(post_install, levels = c(0, 1),
                          labels = c("Pre", "Post")),
    treatment = factor(treatment, levels = c(0, 1),
                       labels = c("Other streets", "PBL installs")),
    outcome = factor(outcome, levels = outcomes) |>
                fct_relabel(\(x) str_replace_all(x, "_", " ") |>
                              str_replace(" per acre", "") |>
                              str_to_sentence())
  )

ggplot(summary_df, aes(x = post_install, y = mean,
                       color = treatment, group = treatment)) +
  geom_point(size = 3) +
  geom_line(linewidth = 0.8) +
  facet_wrap(~outcome, nrow = 1,
             labeller = labeller(outcome = label_wrap_gen(15))) +
  scale_color_manual(values = c("Other streets"   = "grey50",
                                "PBL installs" = "#4575b4")) +
  labs(
    title = "Mean annual crashes per acre before and after bike lane installation",
    x     = NULL,
    y     = "Crashes per acre",
    color = NULL
  ) +
  theme_minimal() +
  theme(
    legend.position    = "bottom",
    panel.grid.major.x = element_blank(),
    strip.text         = element_text(size = 7)
  )
